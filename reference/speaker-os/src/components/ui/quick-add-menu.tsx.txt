// The Quick add menu, opened by the gold hexagon in the middle of the nav.
// One sheet with everything you can add from anywhere: an event (quick or full
// form), a task, a contact or a payment on any event: just the four tiles. Brand: tiles are neutral surfaces with accent icons (no color
// per category), sentence-case labels, one blue main action per form.

type View = 'menu' | 'event' | 'task' | 'contact' | 'payment'

const inputCls =
  'mt-1 block w-full rounded-brand-md border border-control-border bg-surface-sunken px-3 py-2.5 text-[15px] text-ink placeholder:text-ink-muted focus:border-accent transition-colors'
const labelCls = 'block text-label text-ink'
const optionalCls = 'font-normal text-ink-muted'

const OPEN_EVENT = 'speakeros:quick-add'

/** Opens the Quick add menu from anywhere (e.g. Contacts' "+"), optionally straight to one form. */
export function openQuickAdd(view: Exclude<View, 'menu'> | 'menu' = 'menu') {
  window.dispatchEvent(new CustomEvent(OPEN_EVENT, { detail: { view } }))
}

/** For AppShell: listen for openQuickAdd() calls. */
export function onOpenQuickAdd(handler: (view: View) => void): () => void {
  const listener = (e: Event) => handler(((e as CustomEvent).detail?.view as View) ?? 'menu')
  window.addEventListener(OPEN_EVENT, listener)
  return () => window.removeEventListener(OPEN_EVENT, listener)
}

export type QuickAddView = View

export function QuickAddMenu({ open, onClose, initialView = 'menu' }: { open: boolean; onClose: () => void; initialView?: View }) {
  const [view, setView] = useState<View>('menu')
  const [data, setData] = useState<QuickAddData | null>(null)
  const [done, setDone] = useState<{ message: string; href: string | null } | null>(null)
  const sheetRef = useRef<HTMLDivElement>(null)
  const pathname = usePathname()

  // The event you're looking at (if any) is the default for task/contact/payment.
  const currentEventId = pathname.match(/^\/opportunities\/([0-9a-f-]{36})/)?.[1] ?? ''

  // Load the menu's data each time it opens (plain fetch, never a server action on mount).
  useEffect(() => {
    if (!open) return
    let cancelled = false
    fetch('/api/quick-add', { cache: 'no-store' })
      .then((r) => (r.ok ? (r.json() as Promise<QuickAddData>) : null))
      .then((d) => { if (!cancelled && d) setData(d) })
      .catch(() => {})
    return () => { cancelled = true }
  }, [open])

  // Reset to the tiles whenever it closes; close on Escape; focus the sheet on open.
  useEffect(() => {
    if (!open) { setView('menu'); setDone(null); return }
    setView(initialView)
    sheetRef.current?.focus()
    const onKey = (e: KeyboardEvent) => e.key === 'Escape' && onClose()
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [open, onClose, initialView])

  // Close when you navigate somewhere from inside the menu.
  useEffect(() => { onClose() }, [pathname]) // eslint-disable-line react-hooks/exhaustive-deps

  const finish = (message: string, href: string | null) => {
    setDone({ message, href })
    setView('menu')
  }

  return (
    <>
      {/* Dims the page (header included); the nav stays on top so the X can close it. */}
      <div
        aria-hidden="true"
        onClick={onClose}
        className={cn(
          'fixed inset-0 z-[45] bg-band/60 transition-opacity duration-200',
          open ? 'opacity-100' : 'pointer-events-none opacity-0',
        )}
      />
      <div
        ref={sheetRef}
        role="dialog"
        aria-modal="true"
        aria-labelledby="quick-add-title"
        tabIndex={-1}
        // inert keeps the hidden sheet out of the tab order and screen readers.
        {...(!open ? { inert: '' as unknown as boolean } : {})}
        className={cn(
          'fixed inset-x-2 bottom-0 z-[46] mx-auto max-w-lg outline-none transition-transform duration-300 ease-out',
          open ? 'translate-y-0' : 'translate-y-full',
        )}
      >
        {/* Rounded sheet that tucks behind the floating nav (padding clears it). */}
        <div className="max-h-[85dvh] overflow-y-auto rounded-t-[28px] border border-b-0 border-border bg-surface px-5 pb-[calc(104px+env(safe-area-inset-bottom))] pt-3">
          <div className="mx-auto mb-4 h-1 w-10 rounded-full bg-border" aria-hidden="true" />

          {view === 'menu' ? (
            <MenuView
              data={data}
              done={done}
              onPick={(v) => { setDone(null); setView(v) }}
              onClose={onClose}
            />
          ) : (
            <div>
              <button
                type="button"
                onClick={() => setView('menu')}
                className="-ml-1 mb-3 inline-flex min-h-[40px] items-center gap-1.5 rounded-brand-sm px-1 text-label font-semibold text-accent"
              >
                <ArrowLeft aria-hidden="true" className="h-4 w-4" strokeWidth={1.75} />
                Quick add
              </button>
              {view === 'event' && <EventForm onDone={finish} />}
              {view === 'task' && <TaskForm events={data?.events ?? []} defaultEventId={currentEventId} onDone={finish} />}
              {view === 'contact' && <ContactForm events={data?.events ?? []} defaultEventId={currentEventId} onDone={finish} />}
              {view === 'payment' && <PaymentForm events={data?.events ?? []} defaultEventId={currentEventId} onDone={finish} />}
            </div>
          )}
        </div>
      </div>
    </>
  )
}

// ── The tiles ────────────────────────────────────────────────

function MenuView({
  data,
  done,
  onPick,
  onClose,
}: {
  data: QuickAddData | null
  done: { message: string; href: string | null } | null
  onPick: (v: View) => void
  onClose: () => void
}) {
  const today = new Date().toLocaleDateString('en-US', { weekday: 'long', month: 'short', day: 'numeric' })
  const openTasks = data?.openTaskCount ?? 0

  return (
    <div className="space-y-5">
      <div className="flex items-baseline justify-between gap-3">
        <h2 id="quick-add-title" className="text-heading text-ink">Quick add</h2>
        <p className="text-caption text-ink-muted">{today}</p>
      </div>

      {done && (
        <p role="status" className="flex items-center gap-2 rounded-brand-md bg-success-tint px-3 py-2.5 text-[15px] text-success">
          <Check aria-hidden="true" className="h-4 w-4 shrink-0" strokeWidth={2} />
          <span className="min-w-0 flex-1">{done.message}</span>
          {done.href && (
            <Link href={done.href} className="shrink-0 font-semibold underline underline-offset-2">
              View
            </Link>
          )}
        </p>
      )}

      <div className="grid grid-cols-2 gap-3">
        {/* Event: two inner actions, like a tile with its own buttons. */}
        <div className="flex flex-col rounded-brand-lg border border-border bg-surface-sunken p-4">
          <TileHeader icon={CalendarPlus} label="Events" />
          <p className="mt-3 text-title text-ink">New event</p>
          <div className="mt-3 space-y-2">
            <button type="button" onClick={() => onPick('event')} className={tileButtonCls}>
              Quick capture
            </button>
            <Link href="/opportunities/new" onClick={onClose} className={tileButtonCls}>
              Full form
            </Link>
          </div>
        </div>

        <Tile icon={ListTodo} label="Tasks" title="Add a task" sub={openTasks ? `${openTasks} open` : 'For any event'} onClick={() => onPick('task')} />
        <Tile icon={UserPlus} label="Contacts" title="Add a contact" sub="To any event" onClick={() => onPick('contact')} />
        <Tile icon={Banknote} label="Payments" title="Log a payment" sub="Deposit, final, partial" onClick={() => onPick('payment')} />
      </div>

    </div>
  )
}

const tileButtonCls =

type OnDone = (message: string, href: string | null) => void

  return (

function EventPicker({
  id,
  events,
  value,
  onChange,
  error,
}: {
  id: string
  events: QuickAddData['events']
  value: string
  onChange: (v: string) => void
  error?: string | null
}) {
  return (
    <div>
      <label htmlFor={id} className={labelCls}>Event</label>
      <select
        id={id}
        value={value}
        onChange={(e) => onChange(e.target.value)}
        aria-invalid={error ? true : undefined}
        className={inputCls}
      >
        <option value="">Choose an event</option>
        {events.map((e) => (
          <option key={e.id} value={e.id}>
            {e.event_name}{e.event_date ? ` (${formatDate(e.event_date)})` : ''}
          </option>
        ))}
      </select>
      {error && <FieldError>{error}</FieldError>}
    </div>
  )
}

/** Quick capture: event name + organization, saved as a lead (as Quick Add always did). */
function EventForm({ onDone }: { onDone: OnDone }) {
  const [eventName, setEventName] = useState('')
  const [organization, setOrganization] = useState('')
  const [tried, setTried] = useState(false)
  const [error, setError] = useState('')
  const [pending, startTransition] = useTransition()
  const inputRef = useRef<HTMLInputElement>(null)
  const suggestions = useSuggestions()
  const router = useRouter()

  useEffect(() => { setTimeout(() => inputRef.current?.focus(), 50) }, [])

  function submit(e: React.FormEvent) {
    e.preventDefault()
    if (pending) return
    if (!eventName.trim()) { setTried(true); inputRef.current?.focus(); return }
    setError('')
    startTransition(async () => {
      try {
        await createOpportunity({
          event_name: eventName.trim(), contact_person: '', organization: organization.trim(), status: 'lead',
          fee: '', event_date: '', event_end_date: '', opportunity_types: [], venue_name: '', venue_address: '',
          contact_email: '', contact_role: '', contact_phone: '', referral_potential: 'none', notes: '', source: '',
          assigned_to: '', event_timezone: '', audience: '', source_event_id: '',
        })
        invalidateSuggestions()
        router.refresh()
        onDone(`Added ${eventName.trim()} to Events as a lead.`, '/pipeline')
      } catch (err) {
        setError(err instanceof Error ? err.message : 'Failed to save')
      }
    })
  }

  return (
    <form onSubmit={submit} noValidate className="space-y-4">
      <FormHeader icon={CalendarPlus} title="New event" sub="Capture it now, fill in the details later." />
      <div>
        <label className={labelCls}>Event name</label>
        <AutocompleteInput
          inputRef={inputRef}
          value={eventName}
          onChange={setEventName}
          options={eventOptions(suggestions.events)}
          onPick={(option) => {
            setEventName(option.value)
            if (!organization.trim() && option.data?.organization) setOrganization(option.data.organization)
          }}
          aria-label="Event name"
          className={inputCls}
        />
        {tried && !eventName.trim() && <FieldError>Enter the event name to add it.</FieldError>}
      </div>
      <div>
        <label className={labelCls}>Organization <span className={optionalCls}>(optional)</span></label>
        <AutocompleteInput
          value={organization}
          onChange={setOrganization}
          options={organizationOptions(suggestions)}
          aria-label="Organization (optional)"
          className={inputCls}
        />
      </div>
      {error && <FormError>{error}</FormError>}
      <Button type="submit" variant="internal" aria-busy={pending || undefined} className="w-full">
        {pending ? 'Saving…' : 'Add to Events'}
      </Button>
      <Link href="/opportunities/new" className="flex items-center justify-center gap-1.5 text-label font-semibold text-accent hover:underline">
        Use the full form
        <ArrowRight aria-hidden="true" className="h-4 w-4" strokeWidth={1.75} />
      </Link>
    </form>
  )
}

function eventName(events: QuickAddData['events'], id: string) {
  return events.find((e) => e.id === id)?.event_name ?? 'the event'
}

function TaskForm({ events, defaultEventId, onDone }: { events: QuickAddData['events']; defaultEventId: string; onDone: OnDone }) {
  const [eventId, setEventId] = useState(defaultEventId)
  const [title, setTitle] = useState('')
  const [due, setDue] = useState('')
  const [tried, setTried] = useState(false)
  const [error, setError] = useState('')
  const [pending, startTransition] = useTransition()
  const router = useRouter()

  function submit(e: React.FormEvent) {
    e.preventDefault()
    if (pending) return
    if (!title.trim() || !eventId) { setTried(true); return }
    setError('')
    startTransition(async () => {
      const res = await addTask(eventId, { title: title.trim(), due_date: due, notes: '' })
      if (!res.success) { setError(res.error); return }
      router.refresh()
      onDone(`Task added to ${eventName(events, eventId)}.`, `/opportunities/${eventId}`)
    })
  }

  return (
    <form onSubmit={submit} noValidate className="space-y-4">
      <FormHeader icon={ListTodo} title="Add a task" sub="It shows on the event and in My actions." />
      <div>
        <label htmlFor="qa-task-title" className={labelCls}>Task</label>
        <input id="qa-task-title" autoFocus value={title} onChange={(e) => setTitle(e.target.value)} placeholder="Send the AV requirements" className={inputCls} />
        {tried && !title.trim() && <FieldError>Enter what needs doing.</FieldError>}
      </div>
      <EventPicker id="qa-task-event" events={events} value={eventId} onChange={setEventId} error={tried && !eventId ? 'Choose the event this task is for.' : null} />
      <div>
        <label htmlFor="qa-task-due" className={labelCls}>Due date <span className={optionalCls}>(optional)</span></label>
        <input id="qa-task-due" type="date" value={due} onChange={(e) => setDue(e.target.value)} className={inputCls} />
      </div>
      {error && <FormError>{error}</FormError>}
      <Button type="submit" variant="internal" aria-busy={pending || undefined} className="w-full">
        {pending ? 'Saving…' : 'Add task'}
      </Button>
    </form>
  )
}

function ContactForm({ events, defaultEventId, onDone }: { events: QuickAddData['events']; defaultEventId: string; onDone: OnDone }) {
  const [eventId, setEventId] = useState(defaultEventId)
  const [form, setForm] = useState({ name: '', role: '', email: '', phone: '' })
  const [tried, setTried] = useState(false)
  const [error, setError] = useState('')
  const [pending, startTransition] = useTransition()
  const router = useRouter()
  const set = (k: keyof typeof form) => (e: React.ChangeEvent<HTMLInputElement>) => setForm((f) => ({ ...f, [k]: e.target.value }))

  function submit(e: React.FormEvent) {
    e.preventDefault()
    if (pending) return
    if (!form.name.trim() || !eventId) { setTried(true); return }
    setError('')
    startTransition(async () => {
      const res = await addContact(eventId, { ...form, notes: '' })
      if (!res.success) { setError(res.error); return }
      invalidateSuggestions()
      router.refresh()
      onDone(`${form.name.trim()} added to ${eventName(events, eventId)}.`, `/opportunities/${eventId}`)
    })
  }

  return (
    <form onSubmit={submit} noValidate className="space-y-4">
      <FormHeader icon={UserPlus} title="Add a contact" sub="Someone else you're working with on an event." />
      <EventPicker id="qa-contact-event" events={events} value={eventId} onChange={setEventId} error={tried && !eventId ? 'Choose the event this contact belongs to.' : null} />
      <div>
        <label htmlFor="qa-contact-name" className={labelCls}>Name</label>
        <input id="qa-contact-name" value={form.name} onChange={set('name')} autoComplete="off" className={inputCls} />
        {tried && !form.name.trim() && <FieldError>Enter their name.</FieldError>}
      </div>
      <div>
        <label htmlFor="qa-contact-role" className={labelCls}>Role <span className={optionalCls}>(optional)</span></label>
        <input id="qa-contact-role" value={form.role} onChange={set('role')} placeholder="AV lead, assistant, sponsor…" className={inputCls} />
      </div>
      <div className="grid gap-4 sm:grid-cols-2">
        <div>
          <label htmlFor="qa-contact-email" className={labelCls}>Email <span className={optionalCls}>(optional)</span></label>
          <input id="qa-contact-email" type="email" inputMode="email" value={form.email} onChange={set('email')} className={inputCls} />
        </div>
        <div>
          <label htmlFor="qa-contact-phone" className={labelCls}>Phone <span className={optionalCls}>(optional)</span></label>
          <input id="qa-contact-phone" type="tel" inputMode="tel" value={form.phone} onChange={set('phone')} className={inputCls} />
        </div>
      </div>
      {error && <FormError>{error}</FormError>}
      <Button type="submit" variant="internal" aria-busy={pending || undefined} className="w-full">
        {pending ? 'Saving…' : 'Add contact'}
      </Button>
    </form>
  )
}

function PaymentForm({ events, defaultEventId, onDone }: { events: QuickAddData['events']; defaultEventId: string; onDone: OnDone }) {
  const [eventId, setEventId] = useState(defaultEventId)
  const [amount, setAmount] = useState('')
  const [type, setType] = useState<PaymentType>('deposit')
  const [date, setDate] = useState(localDateStr())
  const [tried, setTried] = useState(false)
  const [error, setError] = useState('')
  const [pending, startTransition] = useTransition()
  const router = useRouter()

  const amountOk = amount.trim() !== '' && Number(amount) > 0

  function submit(e: React.FormEvent) {
    e.preventDefault()
    if (pending) return
    if (!amountOk || !eventId || !date) { setTried(true); return }
    setError('')
    startTransition(async () => {
      const res = await addPayment(eventId, { amount, payment_type: type, received_date: date, notes: '' })
      if (!res.success) { setError(res.error); return }
      router.refresh()
      onDone(`Payment logged for ${eventName(events, eventId)}.`, `/opportunities/${eventId}`)
    })
  }

  return (
    <form onSubmit={submit} noValidate className="space-y-4">
      <FormHeader icon={Banknote} title="Log a payment" sub="Money received for an event." />
      <EventPicker id="qa-pay-event" events={events} value={eventId} onChange={setEventId} error={tried && !eventId ? 'Choose the event this payment is for.' : null} />
      <div className="grid grid-cols-2 gap-4">
        <div>
          <label htmlFor="qa-pay-amount" className={labelCls}>Amount</label>
          <input id="qa-pay-amount" type="number" inputMode="decimal" min="0" step="0.01" value={amount} onChange={(e) => setAmount(e.target.value)} placeholder="0.00" className={inputCls} />
          {tried && !amountOk && <FieldError>Enter the amount.</FieldError>}
        </div>
        <div>
          <label htmlFor="qa-pay-date" className={labelCls}>Received</label>
          <input id="qa-pay-date" type="date" value={date} onChange={(e) => setDate(e.target.value)} className={inputCls} />
          {tried && !date && <FieldError>Enter the date.</FieldError>}
        </div>
      </div>
      <div>
        <label htmlFor="qa-pay-type" className={labelCls}>Type</label>
        <select id="qa-pay-type" value={type} onChange={(e) => setType(e.target.value as PaymentType)} className={inputCls}>
          {PAYMENT_TYPES.map((t) => <option key={t.value} value={t.value}>{t.label}</option>)}
        </select>
      </div>
      {error && <FormError>{error}</FormError>}
      <Button type="submit" variant="internal" aria-busy={pending || undefined} className="w-full">
        {pending ? 'Saving…' : 'Log payment'}
      </Button>
    </form>
  )
}

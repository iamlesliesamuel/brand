'use client'

import { useEffect, useState } from 'react'
import { cn } from '@/lib/utils'
import { greetingFor } from './headline-model'

/**
 * The dashboard's one serif headline: today's date in small type above a
 * greeting, the way Apple's apps open. No summary sentence: the cards right
 * below say what needs you, so repeating it in words was redundant. The one
 * exception is a quiet day, when every action card is hidden and the screen
 * would otherwise look empty for no stated reason.
 */
export function DashboardHeadline({
  firstName,
  allClear,
}: {
  firstName: string | null
  /** Nothing waiting: no replies, follow-ups or tasks on the dashboard. */
  allClear: boolean
}) {
  // Date and greeting follow the viewer's clock: computed after mount and
  // faded in, so the server render never disagrees with the browser.
  const [now, setNow] = useState<Date | null>(null)
  useEffect(() => setNow(new Date()), [])

  const fade = cn('transition-opacity duration-300', now ? 'opacity-100' : 'opacity-0')
  const date = now?.toLocaleDateString('en-US', { weekday: 'long', month: 'long', day: 'numeric' })

  return (
    <div>
      <p className={cn('text-label text-ink-muted', fade)}>{date ?? ' '}</p>
      <h1 className={cn('mt-0.5 font-display text-display-md text-ink', fade)}>
        {now ? greetingFor(now.getHours()) : 'Welcome back'}
        {firstName ? `, ${firstName}` : ''}
      </h1>
      {allClear && <p className={cn('mt-1 text-body text-ink-muted', fade)}>You&apos;re all caught up.</p>}
    </div>
  )
}
